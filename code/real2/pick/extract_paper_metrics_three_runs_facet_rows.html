resolve_json_path <- function(roots, method_name, run_idx, alg_dir, json_name) {
  dset <- sprintf("oocyte_zygote_%s_pairs_r%02d", method_name, run_idx)
  # Layout A: <root>/<dset>/1M/<alg>/<json>
  candidates_a <- file.path(roots, dset, "1M", alg_dir, json_name)
  # Layout B: <root>/<alg>/<method>/<dset>_1M_<alg>.json (optional custom naming)
  candidates_b <- file.path(roots, alg_dir, method_name, sprintf("%s_1M_%s", dset, json_name))
  # Layout C: <root>/<alg>/<method>/<json> (single-run copy; not ideal but supported)
  candidates_c <- file.path(roots, alg_dir, method_name, json_name)
  candidates <- c(candidates_a, candidates_b, candidates_c)
  hit <- candidates[file.exists(candidates)]
  if (length(hit) > 0) return(hit[[1]])
  NA_character_
}

4) Extract ARI/NMI/Silhouette Families + AvgBIO from JSON

first_numeric <- function(x) {
  if (is.null(x)) return(NA_real_)
  if (is.list(x)) x <- unlist(x, recursive = TRUE, use.names = FALSE)
  vals <- suppressWarnings(as.numeric(x))
  if (length(vals) == 0 || all(is.na(vals))) return(NA_real_)
  vals[which(!is.na(vals))[1]]
}

extract_relevant_metrics <- function(path) {
  if (is.na(path) || !file.exists(path)) {
    return(tibble(metric = character(), value = numeric(), avgBio = numeric()))
  }

  j <- jsonlite::fromJSON(path, simplifyVector = FALSE)
  keys <- names(j)
  if (is.null(keys) || length(keys) == 0) {
    return(tibble(metric = character(), value = numeric(), avgBio = numeric()))
  }

  vals <- purrr::map_dbl(keys, ~ first_numeric(j[[.x]]))
  base_df <- tibble(metric = keys, value = vals) %>%
    filter(!is.na(value)) %>%
    filter(grepl("(ari|nmi|silhouette)", metric, ignore.case = TRUE)) %>%
    filter(!grepl("silhouette-gt", metric, ignore.case = TRUE))

  best_ari <- first_numeric(j[["best_ari"]])
  best_nmi <- first_numeric(j[["best_nmi"]])
  best_asw <- first_numeric(j[["best_silhouette"]])
  avgBio <- mean(c(best_ari, best_nmi, best_asw), na.rm = TRUE)
  if (all(is.na(c(best_ari, best_nmi, best_asw)))) avgBio <- NA_real_

  bind_rows(
    base_df,
    tibble(metric = "avgBio", value = avgBio)
  )
}

# group separators and labels (same grouping idea as one-panel plot)
group_boundaries <- tibble::tibble(
  metric = levels(plot_df$metric),
  idx = seq_along(levels(plot_df$metric)),
  metric_group = case_when(
    grepl("^ari|best_ari$", metric, ignore.case = TRUE) ~ "ARI",
    grepl("^nmi|best_nmi$", metric, ignore.case = TRUE) ~ "NMI",
    grepl("silhouette", metric, ignore.case = TRUE) ~ "Silhouette",
    metric == "avgBio" ~ "AvgBIO",
    TRUE ~ "Other"
  )
)

separator_x <- group_boundaries %>%
  group_by(metric_group) %>%
  summarise(last_idx = max(idx), .groups = "drop") %>%
  arrange(last_idx)

Avg BIO: average of ARI, NMI, and ASW

resolve_embedding_png <- function(roots, method_name, run_idx, alg_dir) {
  dset <- sprintf("oocyte_zygote_%s_pairs_r%02d", method_name, run_idx)
  candidates <- file.path(roots, dset, "1M", alg_dir, "celltype_plots", "embedding.png")
  hit <- candidates[file.exists(candidates)]
  if (length(hit) > 0) return(hit[[1]])
  NA_character_
}
